smart_relay.ino 
int relayPin = 8;
bool lampState = false;

void setup() {
  pinMode(relayPin, OUTPUT);
  Serial.begin(9600);
}

void loop() {
  if (receiveSignal() == HIGH) {
    digitalWrite(relayPin, HIGH);
    Serial.println("Lampu ON");
  } else {
    digitalWrite(relayPin, LOW);
  }
}
